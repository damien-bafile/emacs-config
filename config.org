  ;; foreground, so matched text stays white on normal rows and dark
  ;; on the lavender selection bar (fixed accent colors wash out on
  ;; lavender). Matches VS Code's command-palette highlighting.
  ;; `completions-common-part' is built in; the orderless faces only
  ;; exist once orderless is loaded, so defer those.
  (set-face-attribute 'completions-common-part nil
                      :weight 'bold :underline nil :foreground 'unspecified)
  (with-eval-after-load 'orderless
    (dolist (face '(orderless-match-face-0 orderless-match-face-1
                    orderless-match-face-2 orderless-match-face-3))
      (set-face-attribute face nil :weight 'bold :underline nil
                          :foreground 'unspecified)))

                        :foreground "#11111b"

#+begin_src emacs-lisp
;; On Windows, make the MSYS2 compiler available so tree-sitter grammars
;; can be built (treesit-auto invokes cc/gcc through exec-path). Prefer
;; the native mingw-w64 gcc; grammars built with the msys gcc segfault
;; Emacs on Windows. On Linux/macOS/BSD the system compiler is already on
;; PATH, so nothing needs to be added.
(when (memq system-type '(ms-dos windows-nt cygwin))
  (let* ((mingw-bins '("C:/msys64/ucrt64/bin" "C:/msys64/mingw64/bin" "C:/msys64/usr/bin"))
         (mingw-bin (seq-find
                     (lambda (d)
                       (file-exists-p (expand-file-name "gcc.exe" d)))
                     mingw-bins)))
    (when mingw-bin
      (add-to-list 'exec-path mingw-bin)
      (setenv "PATH" (concat mingw-bin path-separator (getenv "PATH"))))))

* Markdown & Mermaid
#+begin_src emacs-lisp
;; npm globals live under %APPDATA%\npm on Windows, and GUI Emacs often
;; starts with a PATH that differs from the terminal; add it to exec-path
;; so Emacs can find mmdc.
(when (memq system-type '(ms-dos windows-nt cygwin))
  (let ((npm-bin (expand-file-name "AppData/Roaming/npm" (getenv "HOME"))))
    (when (file-directory-p npm-bin)
      (add-to-list 'exec-path npm-bin))))

;; markdown-mode provides the editing commands; treesit-auto transparently
;; remaps .md files to the built-in markdown-ts-mode once the grammars are
;; built, degrading gracefully to this Elisp mode otherwise.
(use-package markdown-mode
  :mode "\\.md\\'"
  :custom
  ;; Fontify fenced code blocks with their language's major mode, so
  ;; ```mermaid blocks get mermaid-mode highlighting inside the buffer.
  (markdown-fontify-code-blocks-natively t)
  ;; pandoc alone only *highlights* ```mermaid blocks; the Lua filter
  ;; rewrites them to <pre class="mermaid"> and the header file loads
  ;; mermaid.js, so C-c C-c p renders actual diagrams in the browser.
  (markdown-command
   (concat (expand-file-name ".local/bin/pandoc.exe" (getenv "HOME"))
           " --standalone"
           " --lua-filter=" (expand-file-name "mermaid-filter.lua" user-emacs-directory)
           " --include-in-header=" (expand-file-name "mermaid-header.html" user-emacs-directory)
           " --include-in-header=" (expand-file-name "preview-style.html" user-emacs-directory)
           " --metadata title=Preview"))
  :config
  ;; Ensure ```mermaid blocks map to mermaid-mode for native fontification
  ;; (markdown-get-lang-mode would also find it via the "mermaid-mode"
  ;; naming convention, this just makes it explicit).
  (add-to-list 'markdown-code-lang-modes '("mermaid" . mermaid-mode))
  ;; C-c C-o opens the file with the Windows default app for .md
  ;; (markdown-open-command also accepts a string program name).
  (setq markdown-open-command
        (lambda ()
          (unless buffer-file-name
            (user-error "Current buffer is not visiting a file"))

;; Standalone .mmd diagram files: syntax highlighting + C-c C-c compile
